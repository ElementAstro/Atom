cmake_minimum_required(VERSION 3.20)
project(atom-component-test)

# Set C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Include directories
include_directories(.)
include_directories(../..)

# Link directories
link_directories(build)
link_directories(../../build/atom/error)
link_directories(../../build/atom/log)

# Create test executable
add_executable(test_basic_functionality test_basic_functionality.cpp)

# Link libraries
target_link_libraries(test_basic_functionality 
    atom-component
    atom-error
    loguru
    ${CMAKE_CURRENT_SOURCE_DIR}/../../libfmt-11.dll
    ${CMAKE_THREAD_LIBS_INIT}
)

# Set C++20 for the target
target_compile_features(test_basic_functionality PUBLIC cxx_std_20)
