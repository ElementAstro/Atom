cmake_minimum_required(VERSION 3.20)

project(atom_memory_simple.test)

find_package(GTest QUIET)

# Only include the simplified test file
set(TEST_SOURCES test_memory_simple.cpp)

add_executable(${PROJECT_NAME} ${TEST_SOURCES})

# Minimal dependencies - no loguru, no complex libraries
target_link_libraries(${PROJECT_NAME} gtest gtest_main atom-memory atom-error)

# Add Windows system libraries if on Windows (minimal set)
if(WIN32)
    target_link_libraries(${PROJECT_NAME} kernel32)
endif()

# Set C++20 standard
set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD_REQUIRED ON)

# Add include directories
target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/atom
)

# Disable memory tracking to avoid complex initialization
target_compile_definitions(${PROJECT_NAME} PRIVATE
    ATOM_MEMORY_TRACKING_DISABLED=1
)
