# Test Framework CMakeLists.txt
# This file can be used to build the test framework as a standalone executable

cmake_minimum_required(VERSION 3.20)

# Create test framework executable
add_executable(example_test_framework test_framework.cpp)

# Set C++20 standard
target_compile_features(example_test_framework PRIVATE cxx_std_20)

# Platform-specific libraries
if(WIN32)
    # Windows-specific libraries for process management
    target_link_libraries(example_test_framework PRIVATE)
else()
    # Unix-specific libraries
    target_link_libraries(example_test_framework PRIVATE)
endif()

# Compiler-specific options
if(MSVC)
    target_compile_options(example_test_framework PRIVATE /W4)
else()
    target_compile_options(example_test_framework PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Set output directory
set_target_properties(example_test_framework PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/example
)

# Add to examples if building as part of main project
if(TARGET atom-log)  # Check if we're building as part of main project
    # Add to example targets
    list(APPEND ATOM_MODULE_TARGETS example_test_framework)
    set(ATOM_MODULE_TARGETS ${ATOM_MODULE_TARGETS} PARENT_SCOPE)
endif()
